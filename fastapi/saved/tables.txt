CREATE  TABLE public.3dmodels (
  model_id bigint GENERATED BY DEFAULT AS IDENTITY NOT NULL,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  product_id bigint NULL,
  file_format character varying NULL,
  url character varying NULL,
  CONSTRAINT 3dmodels_pkey PRIMARY KEY (model_id),
  CONSTRAINT 3dmodels_product_id_fkey FOREIGN KEY (product_id) REFERENCES product(product_id)
) TABLESPACE pg_default;


create table public.order_items (
  item_id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  order_id bigint null,
  quantity bigint null,
  poduct_id bigint null,
  price character varying null,
  constraint order_items_pkey primary key (item_id),
  constraint order_items_order_id_fkey foreign KEY (order_id) references orders (order_id),
  constraint order_items_poduct_id_fkey foreign KEY (poduct_id) references product (product_id)
) TABLESPACE pg_default;


create table public.orders (
  order_id bigint generated by default as identity not null,
  created_at timestamp with time zone not null default now(),
  order_date timestamp without time zone null,
  status character varying null,
  user_id bigint null,
  constraint orders_pkey primary key (order_id),
  constraint orders_order_id_fkey foreign KEY (order_id) references product (product_id)
) TABLESPACE pg_default;

create table public.product (

create table public.products (
  id bigint generated always as identity not null,
  name text not null,
  description text null,
  price numeric(10, 2) not null,
  category text not null,
  image_url text not null,
  weather text null,
  temp text null,
  event text null,
  rating numeric(3, 1) null default 0,
  embedding public.vector null,

  updated_at timestamp with time zone not null default timezone ('utc'::text, now()),
